cmake_minimum_required(VERSION 3.22)
project(socialtech_simreal)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)
find_package(ament_cmake REQUIRED)
foreach(p rclcpp geometry_msgs nav_msgs sensor_msgs)
  find_package(${p} REQUIRED)
endforeach()
foreach(n sim_base_model sim_wheel_odom sim_scan_model sim_imu_model)
  add_executable(${n} src/${n}.cpp)
  target_include_directories(${n} PRIVATE include)
  ament_target_dependencies(${n} rclcpp geometry_msgs nav_msgs sensor_msgs)
  install(TARGETS ${n} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_models test/test_models.cpp)
  target_include_directories(test_models PRIVATE include)
endif()
ament_package()
