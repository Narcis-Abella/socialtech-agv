cmake_minimum_required(VERSION 3.22)
project(socialtech_costmap)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
add_library(decay_obstacle_layer SHARED src/decay_obstacle_layer.cpp)
ament_target_dependencies(decay_obstacle_layer nav2_costmap_2d pluginlib rclcpp)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS decay_obstacle_layer LIBRARY DESTINATION lib)
ament_export_libraries(decay_obstacle_layer)
ament_package()
