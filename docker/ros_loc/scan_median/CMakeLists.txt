cmake_minimum_required(VERSION 3.8)
project(scan_median)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)
find_package(ament_cmake REQUIRED)
foreach(p rclcpp sensor_msgs tf2 tf2_ros message_filters Eigen3)
  find_package(${p} REQUIRED)
endforeach()
add_executable(scan_median_node src/scan_median_node.cpp)
target_include_directories(scan_median_node PRIVATE include)
target_link_libraries(scan_median_node Eigen3::Eigen)
ament_target_dependencies(scan_median_node rclcpp sensor_msgs tf2 tf2_ros message_filters)
install(TARGETS scan_median_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_binner test/test_binner.cpp)
  target_include_directories(test_binner PRIVATE include)
  target_compile_definitions(test_binner PRIVATE GOLDEN_PATH="${CMAKE_CURRENT_SOURCE_DIR}/test/golden.txt")
endif()
ament_package()
