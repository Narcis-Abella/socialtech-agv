cmake_minimum_required(VERSION 3.22)
project(socialtech_mapping)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# -ffp-contract=off: no fused multiply-add (GCC fuses by default on aarch64), so the float64 results follow the numpy arithmetic of tools/slam_bench/beams_oracle.py
add_compile_options(-Wall -Wextra -ffp-contract=off)
find_package(ament_cmake REQUIRED)
find_package(OpenMP REQUIRED)
foreach(p rclcpp rosbag2_cpp rosbag2_storage sensor_msgs)
  find_package(${p} REQUIRED)
endforeach()
add_library(socialtech_mapping_beams SHARED src/beams.cpp)
target_include_directories(socialtech_mapping_beams PRIVATE include)
target_link_libraries(socialtech_mapping_beams OpenMP::OpenMP_CXX)
ament_target_dependencies(socialtech_mapping_beams rclcpp rosbag2_cpp rosbag2_storage sensor_msgs)
install(TARGETS socialtech_mapping_beams LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_beams test/test_beams.cpp)
  target_include_directories(test_beams PRIVATE include)
endif()
ament_package()
