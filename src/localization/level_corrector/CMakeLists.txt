cmake_minimum_required(VERSION 3.8)
project(level_corrector)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
# small_gicp is not an apt package: the image builds the pinned commit (fa0cfc9, the one register_v2 uses) and passes -DCMAKE_PREFIX_PATH to it.
find_package(small_gicp REQUIRED)
foreach(p rclcpp geometry_msgs nav_msgs sensor_msgs std_msgs tf2 tf2_ros tf2_eigen)
  find_package(${p} REQUIRED)
endforeach()
add_executable(level_corrector_node src/level_corrector_node.cpp)
target_include_directories(level_corrector_node PRIVATE include)
target_link_libraries(level_corrector_node Eigen3::Eigen small_gicp::small_gicp)
ament_target_dependencies(level_corrector_node rclcpp geometry_msgs nav_msgs sensor_msgs std_msgs tf2 tf2_ros tf2_eigen)
install(TARGETS level_corrector_node DESTINATION lib/${PROJECT_NAME})
# test/test_level_corrector_node.py runs against the installed node: python3 test_level_corrector_node.py
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_core test/test_core.cpp)
  target_include_directories(test_core PRIVATE include)
  target_link_libraries(test_core Eigen3::Eigen)
  ament_add_gtest(test_map_io test/test_map_io.cpp)
  target_include_directories(test_map_io PRIVATE include)
  target_link_libraries(test_map_io Eigen3::Eigen)
  ament_add_gtest(test_registration test/test_registration.cpp)
  target_include_directories(test_registration PRIVATE include)
  target_link_libraries(test_registration Eigen3::Eigen small_gicp::small_gicp)
endif()
ament_package()
