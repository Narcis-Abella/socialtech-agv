cmake_minimum_required(VERSION 3.8)
project(gn_loc)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)
find_package(ament_cmake REQUIRED)
foreach(p rclcpp rclcpp_lifecycle bondcpp geometry_msgs nav_msgs sensor_msgs std_msgs tf2 tf2_ros message_filters Eigen3)
  find_package(${p} REQUIRED)
endforeach()
add_executable(gn_loc_node src/gn_loc_node.cpp)
target_include_directories(gn_loc_node PRIVATE include)
target_link_libraries(gn_loc_node Eigen3::Eigen)
ament_target_dependencies(gn_loc_node rclcpp rclcpp_lifecycle bondcpp geometry_msgs nav_msgs sensor_msgs std_msgs tf2 tf2_ros message_filters)
# offline runner of the core over a replay bag (study tool, not used by the node)
find_package(rosbag2_cpp REQUIRED)
find_package(nav2_map_server REQUIRED)
find_package(tf2_msgs REQUIRED)
add_executable(gn_offline src/gn_offline.cpp)
target_include_directories(gn_offline PRIVATE include)
target_link_libraries(gn_offline Eigen3::Eigen)
ament_target_dependencies(gn_offline rclcpp rosbag2_cpp nav2_map_server sensor_msgs tf2_msgs nav_msgs)
install(TARGETS gn_loc_node gn_offline DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_gn_core test/test_gn_core.cpp)
  target_include_directories(test_gn_core PRIVATE include)
  target_link_libraries(test_gn_core Eigen3::Eigen)
endif()
ament_package()
