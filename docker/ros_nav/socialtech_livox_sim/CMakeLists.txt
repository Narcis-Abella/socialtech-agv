cmake_minimum_required(VERSION 3.22)
project(socialtech_livox_sim)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)
find_package(ament_cmake REQUIRED)
foreach(p rclcpp nav_msgs sensor_msgs livox_ros_driver2)
  find_package(${p} REQUIRED)
endforeach()
find_package(embree 4 REQUIRED)
add_executable(livox_mid360_sim src/livox_mid360_sim.cpp)
target_include_directories(livox_mid360_sim PRIVATE include)
target_link_libraries(livox_mid360_sim embree)
ament_target_dependencies(livox_mid360_sim rclcpp nav_msgs sensor_msgs livox_ros_driver2)
install(TARGETS livox_mid360_sim DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_caster test/test_caster.cpp)
  target_include_directories(test_caster PRIVATE include)
  target_link_libraries(test_caster embree)
endif()
ament_package()
